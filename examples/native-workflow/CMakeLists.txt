cmake_minimum_required(VERSION 3.25)
project(crown_native_workflow_example LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Crown 0.1 CONFIG REQUIRED)
add_executable(crown_native_workflow main.cpp)
target_link_libraries(crown_native_workflow PRIVATE CROWN::explain)
target_compile_options(crown_native_workflow PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
